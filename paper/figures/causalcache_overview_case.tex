  \fi

      (#1-0.19,#2-0.26) rectangle (#1+0.19,#2+0.26);

    \draw[accO, very thick, rounded corners=1.2pt, fill=fillO]
      (#1-0.19,#2-0.26) rectangle (#1+0.19,#2+0.26);
  \fi
  \fill[chipgray] (#1-0.13,#2-0.20) rectangle (#1+0.13,#2-0.10);
}
